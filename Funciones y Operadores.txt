-- Ejercicio 1: Concatenacion de Nombre Completo

-- Ejercicio 3: Fecha de Vencimiento de la Tarjeta de Credito
------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------
SELECT
    tarjeta_id,
    cliente_id,
    numero_tarjeta,
    fecha_vencimiento,
    DATEDIFF(day, GETDATE(), fecha_vencimiento) AS "Dias para Vencimiento" -- Calculamos la diferencia en días hasta la fecha de vencimiento.

    [cli].[tarjetas_credito];

-- Ejercicio 6: Precio Promedio por Categoria
------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------
SELECT
    c.categoria,
    AVG(p.precio) AS "Precio Promedio" -- Calculamos el precio promedio de los productos.
FROM
    [sell].[productos] p
INNER JOIN
    [sell].[categoria] c ON p.categoria_id = c.categoria_id -- Unimos las tablas por 'categoria_id'.
GROUP BY
    c.categoria_id, c.categoria -- Agrupamos por categoría para calcular el promedio por cada una.
ORDER BY
    c.categoria; -- Ordenamos por categoría.

-- Ejercicio 9: Número de Ventas por Cliente
------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------
SELECT
    c.cliente_id,
    c.nombre,
    c.apellido,
    COUNT(v.venta_id) AS "Número de Ventas" -- Contamos el número de ventas por cliente.

INNER JOIN

-- Ejercicio 10: Cantidad de Productos por Cliente en Carrito
------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------
SELECT
    c.cliente_id,
    c.nombre,
    c.apellido,
    SUM(ISNULL(dcc.cantidad, 0)) AS "Productos en Carrito" -- Sumamos la cantidad de productos en el carrito (tratando nulos como 0).
FROM
    [cli].[clientes] c
LEFT JOIN
    [sell].[carrito_compras] cc ON c.cliente_id = cc.cliente_id AND cc.abandonado = 0 -- Unimos por 'cliente_id' para carritos no abandonados.
LEFT JOIN
    [sell].[detalle_carrito_compras] dcc ON cc.carrito_id = dcc.carrito_id -- Unimos por 'carrito_id' para obtener los detalles.
GROUP BY
    c.cliente_id, c.nombre, c.apellido -- Agrupamos por cliente.
ORDER BY
    c.cliente_id; -- Ordenamos por 'cliente_id'.

-- Ejercicio 11: Total de Ventas por Mes
------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------
SELECT
    YEAR(fecha_venta) AS Anio, -- Extraemos el año de la fecha de venta.
    DATENAME(month, fecha_venta) AS Mes, -- Obtenemos el nombre del mes de la fecha de venta.
    MONTH(fecha_venta) AS MesNumero, -- Obtenemos el número del mes para ordenar.
    SUM(total_venta) AS "Total Ventas" -- Sumamos el total de ventas.
FROM
    [sell].[ventas]
GROUP BY
    YEAR(fecha_venta),
    MONTH(fecha_venta),
    DATENAME(month, fecha_venta) -- Agrupamos por año y mes.
ORDER BY
    Anio,
    MesNumero; -- Ordenamos por año y número de mes.

-- Ejercicio 13: Cantidad de Productos Vendidos por Categoría
------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------
SELECT
    c.categoria,
    SUM(dv.cantidad) AS "Productos Vendidos" -- Sumamos la cantidad de productos vendidos.
FROM
    [sell].[detalle_ventas] dv
INNER JOIN
    [sell].[productos] p ON dv.producto_id = p.producto_id -- Unimos por 'producto_id'.
INNER JOIN
    [sell].[categoria] c ON p.categoria_id = c.categoria_id -- Unimos por 'categoria_id'.
GROUP BY
    c.categoria_id, c.categoria -- Agrupamos por categoría.
ORDER BY
    c.categoria; -- Ordenamos por categoría.

-- Ejercicio 14: Tarjeta de Crédito Más Antigua
------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------
SELECT
    c.cliente_id,
    c.nombre,
    c.apellido,
    MIN(tc.fecha_vencimiento) AS "Tarjeta Crédito Antigua" -- Obtenemos la fecha de vencimiento más antigua.
FROM
    [cli].[clientes] c
INNER JOIN
    [cli].[tarjetas_credito] tc ON c.cliente_id = tc.cliente_id -- Unimos por 'cliente_id'.
GROUP BY
    c.cliente_id, c.nombre, c.apellido -- Agrupamos por cliente.
ORDER BY
    c.cliente_id; -- Ordenamos por 'cliente_id'.

-- Ejercicio 15: Precio Total de Productos en Carrito de Compras
------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------
SELECT
    c.cliente_id,
    c.nombre,
    c.apellido,
    SUM(ISNULL(dcc.cantidad * p.precio, 0)) AS "Precio Total Carrito" -- Calculamos el precio total de los productos en el carrito.
FROM
    [cli].[clientes] c
LEFT JOIN
    [sell].[carrito_compras] cc ON c.cliente_id = cc.cliente_id AND cc.abandonado = 0 -- Unimos por 'cliente_id' para carritos no abandonados.
LEFT JOIN
    [sell].[detalle_carrito_compras] dcc ON cc.carrito_id = dcc.carrito_id -- Unimos por 'carrito_id'.
LEFT JOIN
    [sell].[productos] p ON dcc.producto_id = p.producto_id -- Unimos por 'producto_id'.
GROUP BY
    c.cliente_id, c.nombre, c.apellido -- Agrupamos por cliente.
ORDER BY
    c.cliente_id; -- Ordenamos por 'cliente_id'.

------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------
-- Ejercicio 16: Cantidad de Clientes por Dirección
------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------
SELECT
    direccion,
    COUNT(cliente_id) AS "Clientes por Dirección" -- Contamos el número de clientes por dirección.
FROM
    [cli].[clientes]
WHERE
    direccion IS NOT NULL AND direccion <> '' -- Excluimos direcciones nulas o vacías.
GROUP BY
    direccion -- Agrupamos por dirección.
ORDER BY
    "Clientes por Dirección" DESC, direccion; -- Ordenamos por cantidad de clientes (descendente) y luego por dirección.

------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------
-- Ejercicio 17: Productos con Precio Superior al Promedio
------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------
SELECT
    producto_id,
    nombre_producto,
    precio AS "Productos Precio Superior Promedio" -- Mostramos los productos con precio superior al promedio.
FROM
    [sell].[productos]
WHERE
    precio > (SELECT AVG(precio) FROM [sell].[productos]) -- Filtramos los productos con precio mayor al promedio (subconsulta).
ORDER BY
    precio DESC; -- Ordenamos por precio descendente.
